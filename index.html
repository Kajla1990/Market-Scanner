// यह एरे कोटक API से आने वाले या ऐतिहासिक 15-मिनट के क्लोजिंग प्राइसेस को स्टोर करेगा let priceHistory = []; const PERIOD = 14; // मूविंग एवरेज या इंडिकेटर पीरियड // कोटक API से जब भी नया डेटा या टिक मिले, यह फंक्शन कॉल होगा function processMarketData(currentPrice) { priceHistory.push(currentPrice); // अगर हमारे पास पर्याप्त डेटा नहीं है, तो इंतजार करें if (priceHistory.length < PERIOD) { document.getElementById("signalDisplay").innerText = "Gathering Data for 15-Min Analysis..."; return; } // केवल आखिरी 15-30 मिनट के ट्रेंड को कैलकुलेट करें (Simple Moving Average - SMA उदाहरण) let sum = 0; for (let i = priceHistory.length - PERIOD; i < priceHistory.length; i++) { sum += priceHistory[i]; } let sma = sum / PERIOD; // सिग्नल तय करना (जो केवल एक निश्चित अंतराल या नई कैंडल पर बदलेगा) const signalBox = document.getElementById("signalDisplay"); if (currentPrice > sma) { signalBox.className = "signal-box bullish"; signalBox.innerHTML = "BULLISH 🚀 (15-Min Trend Up)"; } else if (currentPrice < sma) { signalBox.className = "signal-box bearish"; signalBox.innerHTML = "BEARISH 📉 (15-Min Trend Down)"; } else { signalBox.className = "signal-box neutral"; signalBox.innerHTML = "NEUTRAL ⚖️"; } } // कैंडलस्टिक पैटर्न न दिखने की समस्या के लिए: // यदि आप खुद का कैनवास (HTML5 Canvas) या चार्ट बना रहे हैं, तो सुनिश्चित करें कि // आप कोटक API के 'Open, High, Low, Close (OHLC)' डाटा को सही अरे (Array) में पास कर रहे हैं।
